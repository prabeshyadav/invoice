# Invoice Management System — runtime dependencies
# Python >=3.12 required (Django 6.1 needs >=3.12; Dockerfile uses
# python:3.12-slim, tested locally on Python 3.14.8).
# Install with: pip install -r requirements.txt
# Only direct dependencies are pinned below; transitive dependencies
# (kombu, billiard, reportlab, Pillow, etc.) are resolved by pip.

# Web framework — invoice/settings.py, invoice/urls.py, invoice/wsgi.py,
# invoice/asgi.py, manage.py, management/ (models, views, forms, admin).
Django==6.1.1

# Distributed task queue + beat scheduler — invoice/celery.py,
# invoice/settings.py (CELERY_BROKER_URL, CELERY_BEAT_SCHEDULE),
# management/tasks.py, docker-compose.yml (celery_worker, celery_beat).
# The [redis] extra declares the kombu redis-transport dependency;
# the client itself is pinned explicitly below.
celery[redis]==5.6.3

# Redis client — Celery broker transport for redis:// URLs
# (CELERY_BROKER_URL, docker-compose.yml redis service).
# Without this, kombu.transport.redis fails at worker startup with
# `AttributeError: 'NoneType' object has no attribute 'Redis'`.
# Version constrained by kombu 5.6: redis>=4.5.2,<6.5 (!=4.5.5, !=5.0.2).
redis==6.4.0

# Invoice PDF rendering — management/views.py (`from xhtml2pdf import pisa`).
xhtml2pdf==0.2.17

# Production WSGI server — Dockerfile CMD
# (`gunicorn --bind 0.0.0.0:8000 invoice.wsgi:application`).
gunicorn==26.2.0

# REST API — management/serializers.py, management/api.py
# (JWT auth via /api/auth/token/).
djangorestframework==3.18.3
djangorestframework-simplejwt==5.5.1
